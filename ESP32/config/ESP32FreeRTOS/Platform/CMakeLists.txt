register_fprime_config(
        ESP32FreeRTOSPlatformTypes
    AUTOCODER_INPUTS
        "${CMAKE_CURRENT_LIST_DIR}/PlatformTypes.fpp"
    HEADERS
        "${CMAKE_CURRENT_LIST_DIR}/PlatformTypes.h"
    CHOOSES_IMPLEMENTATIONS
        Os_File_Stub
        Os_Console_FreeRTOS
        Os_Task_FreeRTOS
        Os_Mutex_FreeRTOS
        Os_Queue_FreeRTOS
        Os_RawTime_FreeRTOS
        Os_Cpu_Stub
        Os_Memory_Stub
        Fw_StringFormat_snprintf
    INTERFACE
    BASE_CONFIG
)

include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/esp32_idf_wrapper.cmake")
esp32_prepare_idf_wrapper()

# ESP-IDF include locations needed by ESP32FreeRTOS modules.
if(NOT DEFINED IDF_PATH)
    if(DEFINED ENV{IDF_PATH})
        set(IDF_PATH "$ENV{IDF_PATH}")
    elseif(DEFINED ENV{ESP32_IDF_PATH})
        set(IDF_PATH "$ENV{ESP32_IDF_PATH}")
    endif()
endif()

if(DEFINED IDF_PATH)
    set(ESP32_IDF_INCLUDE_DIRS
        "${IDF_PATH}/components/xtensa/esp32/include"
        "${IDF_PATH}/components/xtensa/include"
        "${IDF_PATH}/components/xtensa/deprecated_include"
        "${IDF_PATH}/components/freertos/FreeRTOS-Kernel/include"
        "${IDF_PATH}/components/freertos/FreeRTOS-Kernel-SMP/include"
        "${IDF_PATH}/components/freertos/FreeRTOS-Kernel/portable/xtensa/include"
        "${IDF_PATH}/components/freertos/FreeRTOS-Kernel/portable/xtensa/include/freertos"
        "${IDF_PATH}/components/freertos/config/include"
        "${IDF_PATH}/components/freertos/config/include/freertos"
        "${IDF_PATH}/components/freertos/config/xtensa/include"
        "${IDF_PATH}/components/freertos/config/xtensa/include/freertos"
        "${IDF_PATH}/components/freertos/esp_additions/include"
        "${IDF_PATH}/components/esp_timer/include"
        "${IDF_PATH}/components/esp_common/include"
        "${IDF_PATH}/components/esp_system/include"
        "${IDF_PATH}/components/esp_driver_uart/include"
        "${IDF_PATH}/components/lwip/include"
        "${IDF_PATH}/components/lwip/include/apps"
        "${IDF_PATH}/components/lwip/include/apps/sntp"
        "${IDF_PATH}/components/lwip/lwip/src/include"
    )

    # Optional include list extracted from an IDF wrapper compile_commands.json.
    set(ESP32_IDF_EXTRA_INCLUDE_DIRS)
    set(ESP32_IDF_EXTRA_INCLUDE_DIRS_FILE_CANDIDATE "")
    if(DEFINED ENV{ESP32_IDF_EXTRA_INCLUDE_DIRS_FILE} AND EXISTS "$ENV{ESP32_IDF_EXTRA_INCLUDE_DIRS_FILE}")
        set(ESP32_IDF_EXTRA_INCLUDE_DIRS_FILE_CANDIDATE "$ENV{ESP32_IDF_EXTRA_INCLUDE_DIRS_FILE}")
    elseif(EXISTS "${CMAKE_SOURCE_DIR}/idf-wrapper/build/fprime_idf_include_dirs.cmake")
        set(ESP32_IDF_EXTRA_INCLUDE_DIRS_FILE_CANDIDATE "${CMAKE_SOURCE_DIR}/idf-wrapper/build/fprime_idf_include_dirs.cmake")
    endif()

    if(ESP32_IDF_EXTRA_INCLUDE_DIRS_FILE_CANDIDATE)
        include("${ESP32_IDF_EXTRA_INCLUDE_DIRS_FILE_CANDIDATE}")
    endif()
    list(APPEND ESP32_IDF_INCLUDE_DIRS ${ESP32_IDF_EXTRA_INCLUDE_DIRS})
    list(FILTER ESP32_IDF_INCLUDE_DIRS EXCLUDE REGEX ".*/linux/.*")
    list(REMOVE_DUPLICATES ESP32_IDF_INCLUDE_DIRS)

    foreach(ESP32_IDF_INCLUDE_DIR IN LISTS ESP32_IDF_INCLUDE_DIRS)
        if(EXISTS "${ESP32_IDF_INCLUDE_DIR}")
            target_include_directories(ESP32FreeRTOSPlatformTypes INTERFACE "${ESP32_IDF_INCLUDE_DIR}")
        endif()
    endforeach()

    # SDK configuration header (sdkconfig.h) is generated by an ESP-IDF configure step.
    set(ESP32_SDKCONFIG_INCLUDE_DIR_CANDIDATE "")
    if(DEFINED ESP32_SDKCONFIG_INCLUDE_DIR)
        set(ESP32_SDKCONFIG_INCLUDE_DIR_CANDIDATE "${ESP32_SDKCONFIG_INCLUDE_DIR}")
    elseif(DEFINED ENV{ESP32_SDKCONFIG_INCLUDE_DIR})
        set(ESP32_SDKCONFIG_INCLUDE_DIR_CANDIDATE "$ENV{ESP32_SDKCONFIG_INCLUDE_DIR}")
    elseif(EXISTS "${CMAKE_SOURCE_DIR}/idf-wrapper/build/config/sdkconfig.h")
        set(ESP32_SDKCONFIG_INCLUDE_DIR_CANDIDATE "${CMAKE_SOURCE_DIR}/idf-wrapper/build/config")
    endif()

    if(ESP32_SDKCONFIG_INCLUDE_DIR_CANDIDATE AND EXISTS "${ESP32_SDKCONFIG_INCLUDE_DIR_CANDIDATE}/sdkconfig.h")
        target_include_directories(ESP32FreeRTOSPlatformTypes INTERFACE "${ESP32_SDKCONFIG_INCLUDE_DIR_CANDIDATE}")
    endif()

    # Optional link information extracted from the IDF wrapper build.ninja.
    set(ESP32_IDF_EXTRA_LINK_DIRECTORIES)
    set(ESP32_IDF_EXTRA_LINK_OPTIONS)
    set(ESP32_IDF_EXTRA_LINK_LIBRARIES)
    set(ESP32_IDF_EXTRA_LINK_INFO_FILE_CANDIDATE "")
    if(DEFINED ENV{ESP32_IDF_EXTRA_LINK_INFO_FILE} AND EXISTS "$ENV{ESP32_IDF_EXTRA_LINK_INFO_FILE}")
        set(ESP32_IDF_EXTRA_LINK_INFO_FILE_CANDIDATE "$ENV{ESP32_IDF_EXTRA_LINK_INFO_FILE}")
    elseif(EXISTS "${CMAKE_SOURCE_DIR}/idf-wrapper/build/fprime_idf_link_info.cmake")
        set(ESP32_IDF_EXTRA_LINK_INFO_FILE_CANDIDATE "${CMAKE_SOURCE_DIR}/idf-wrapper/build/fprime_idf_link_info.cmake")
    endif()
    if(ESP32_IDF_EXTRA_LINK_INFO_FILE_CANDIDATE)
        include("${ESP32_IDF_EXTRA_LINK_INFO_FILE_CANDIDATE}")
    endif()

    if(ESP32_IDF_EXTRA_LINK_DIRECTORIES)
        target_link_directories(ESP32FreeRTOSPlatformTypes INTERFACE ${ESP32_IDF_EXTRA_LINK_DIRECTORIES})
    endif()
    if(ESP32_IDF_EXTRA_LINK_OPTIONS)
        target_link_options(ESP32FreeRTOSPlatformTypes INTERFACE ${ESP32_IDF_EXTRA_LINK_OPTIONS})
    endif()
    if(ESP32_IDF_EXTRA_LINK_LIBRARIES)
        target_link_libraries(ESP32FreeRTOSPlatformTypes INTERFACE
            "-Wl,--start-group"
            ${ESP32_IDF_EXTRA_LINK_LIBRARIES}
            "-Wl,--end-group"
        )
    endif()

    # Match key ESP-IDF C++ compile settings to avoid pulling unsupported
    # libstdc++ startup paths on ESP32.
    target_compile_options(ESP32FreeRTOSPlatformTypes INTERFACE
        "$<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>"
        "$<$<COMPILE_LANGUAGE:CXX>:-fno-rtti>"
        "$<$<COMPILE_LANGUAGE:CXX>:-fuse-cxa-atexit>"
    )
    target_compile_definitions(ESP32FreeRTOSPlatformTypes INTERFACE
        ESP_PLATFORM
        _GLIBCXX_HAVE_POSIX_SEMAPHORE
        _GLIBCXX_USE_POSIX_SEMAPHORE
        _GNU_SOURCE
        _POSIX_READER_WRITER_LOCKS
    )
endif()
